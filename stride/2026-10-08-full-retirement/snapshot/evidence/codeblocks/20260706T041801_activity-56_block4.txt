# Helper block guaranteeing metadata immutability across transfer gates
def freeze_structure(data: Dict[str, Any]) -> Dict[str, Any]:
    """Recursively processes mutable input maps into deeply frozen dictionaries."""
    return {k: freeze_structure(v) if isinstance(v, dict) else v for k, v in data.items()}

@dataclass(frozen=True)
class ContextEnvelope:
    """Immutable data container protecting payloads and cryptographic history paths."""
    payload_data: Dict[str, Any]
    session_state_mapping: Dict[str, Any]
    header_mapping: Dict[str, Any]
    status_string: str = "INITIALIZED"

class ComposableSystemModule(Protocol):
    """Unified asynchronous protocol layout required for all system nodes."""
    async def process_payload(self, context_envelope: ContextEnvelope) -> ContextEnvelope:
        ...

class MAGNA_Orchestrator:
    """
    Central logic controller for system authentication, handshake validation,
    and transaction ledger tracking tracking across execution components.
    """
    def __init__(self):
        self.module_registry: Dict[str, Any] = {}
        self.ledger: List[str] = []

    def register_as_module(self, name: str):
        """Active decorator handling system registration tracking maps."""
        def decorator(cls):
            cls.module_name = name
            self.module_registry[name] = cls
            return cls
        return decorator

    def _generate_hash(self, data: str) -> str:
        """Generates explicit SHA-256 verification signatures."""
        return hashlib.sha256(data.encode("utf-8")).hexdigest()

    def handshake(self, source: str, destination: str, payload: str) -> dict:
        """Validates systemic tracking credentials across components."""
        if source not in self.module_registry or destination not in self.module_registry:
            return {
                "status": "denied",
                "reason": f"Handshake validation failed: Unregistered component(s) [{source} -> {destination}]"
            }

        timestamp = datetime.now().isoformat()
        handshake_id = self._generate_hash(f"{source}{destination}{payload}{timestamp}")
        return {"status": "authorized", "tid": timestamp, "hash": handshake_id}

    def transfer(self, source: str, destination: str, data: str) -> bool:
        """Executes secure, audited transactions across authorized component lanes."""
        auth = self.handshake(source, destination, data)
        if auth["status"] == "authorized":
            log_entry = f"[{auth['tid']}] {source} -> {destination} | Hash: {auth['hash']}"
            self.ledger.append(log_entry)
            return True
        return False

# Instantiate the active system orchestration engine
MAGNA = MAGNA_Orchestrator()

class GsaUniversalAdapter:
    """
    Wraps architectural blocks to enforce cryptographic history sequence verifications.
    """
    def __init__(self, underlying_module: Any) -> None:
        self.module = underlying_module
        self.actor_name = type(underlying_module).__name__

    async def process_payload(self, context_envelope: ContextEnvelope) -> ContextEnvelope:
        """Validates tracking ledgers across boundaries before releasing internal components."""
        headers = dict(context_envelope.header_mapping)
        hash_history = list(headers.get("gsa_chain_history", []))
        current_iteration = headers.get("gsa_loop_iteration", 0)
        
        upstream_hash = hash_history[-1] if hash_history else "GENESIS_ANCHOR"
        
        # Invoke matching logic pattern on target module execution
        if hasattr(self.module, "execute_aggregation"):
            # Refactored for structural compatibility with text aggregators
            raw_tx = context_envelope.payload_data.get("raw_transcript", "")
            src = context_envelope.payload_data.get("source_node", "")
            dst = context_envelope.payload_data.get("dest_node", "")
            result = self.module.execute_aggregation(src, dst, raw_tx)
            
            output_envelope = ContextEnvelope(
                payload_data={"aggregated_payload": result},
                session_state_mapping=dict(context_envelope.session_state_mapping),
                header_mapping=dict(context_envelope.header_mapping)
            )
        else:
            output_envelope = context_envelope

        updated_headers = dict(output_envelope.header_mapping)
        next_iteration = current_iteration + 1
        
        # Calculate next block signature tracking element
        serialized_payload = json.dumps(output_envelope.payload_data, sort_keys=True)
        buffer_source = f"parent:{upstream_hash}||iter:{next_iteration}||payload:{serialized_payload}"
        outbound_hash = hashlib.sha256(buffer_source.encode("utf-8")).hexdigest()
        
        hash_history.append(outbound_hash)
        updated_headers["gsa_chain_history"] = hash_history
        updated_headers["gsa_loop_iteration"] = next_iteration
        updated_headers["gsa_interlock_hash"] = outbound_hash
        updated_headers["gsa_last_actor"] = self.actor_name

        return replace(output_envelope, header_mapping=freeze_structure(updated_headers))

class GovernanceEngine:
    """
    Ingress safety layer handling pronominal sanitization and logic loop guards.
    """
    REGEX_IDENTITY: Final = re.compile(r"\b(i|me|my|mine|myself|we|us|our|ours|ourselves)\b", re.IGNORECASE)
    KINETIC_THROTTLE: Final = 0.815
    FORBIDDEN_PATTERNS: Final = ["paradox", "recursion"]

    def __init__(self):
        self.seen_outputs: set[str] = set()

    def sanitize_payload(self, text: str) -> str:
        """Purges pronominal markers to preserve objective context tracking."""
        return self.REGEX_IDENTITY.sub("[SUBJECT_BETA]", text).strip()

    async def verify_and_pace(self, text: str) -> bool:
        """Verifies context loop patterns and applies necessary loop pacing delays."""
        if any(p in text.lower() for p in self.FORBIDDEN_PATTERNS):
            return False
        await asyncio.sleep(self.KINETIC_THROTTLE)
        return True

@MAGNA.register_as_module("ChatAggregator")
class ChatAggregator:
    """Pulls text segments from raw streaming buffers, extracting target code structures."""
    def __init__(self, orchestrator: MAGNA_Orchestrator):
        self.orchestrator = orchestrator

    def extract_code_blocks(self, raw_text: str) -> List[str]:
        """Runs sequential regex evaluations to separate markdown block components."""
        pattern = r"```python(.*?)```"
        return [block.strip() for block in re.findall(pattern, raw_text, re.DOTALL)]

    def execute_aggregation(self, source_node: str, dest_node: str, raw_transcript: str) -> Optional[str]:
        """Consolidates text outputs and transmits packets via active handshake steps."""
        code_blocks = self.extract_code_blocks(raw_transcript)
        total_blocks = len(code_blocks)
        combined_code = "\n\n# --- Unified Code Block ---\n\n".join(code_blocks)
        total_lines = len(combined_code.splitlines()) if combined_code else 0
        
        metadata_header = (
            f"# ==========================================\n"
            f"# METADATA SUMMARY\n"
            f"# Total Code Blocks Extracted: {total_blocks}\n"
            f"# Total Aggregated Code Lines: {total_lines}\n"
            f"# Aggregation Timestamp: {datetime.now().isoformat()}\n"
            f"# ==========================================\n\n"
        )
        
        combined_payload = metadata_header + (combined_code if combined_code else f"# Compiled Prose\n{raw_transcript[:100]}...")
        success = self.orchestrator.transfer(source_node, dest_node, combined_payload)
        return combined_payload if success else None
